## 一、日志系统简介

### 1.1 为什么需要日志？

- ✅ 问题排查：记录错误信息，快速定位问题
- ✅ 性能监控：记录接口响应时间
- ✅ 安全审计：记录用户操作
- ✅ 业务分析：统计访问量、用户行为

### 1.2 Python 标准库 logging

```python
import logging

# 配置日志
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    filename='app.log'
)

logger = logging.getLogger(__name__)

# 使用日志
logger.debug("调试信息")
logger.info("普通信息")
logger.warning("警告信息")
logger.error("错误信息")
logger.critical("严重错误")
```

## 三、在 FastAPI 中集成日志

### 3.1 创建日志配置文件

```python
"""
日志配置模块
"""
import sys
from loguru import logger
from pathlib import Path
from typing import Optional


def setup_logging(
    log_level: str = "INFO",
    log_file: Optional[str] = None,
    rotation: str = "10 MB",
    retention: str = "7 days",
    json_format: bool = False
):
    """
    配置日志系统
    
    Args:
        log_level: 日志级别
        log_file: 日志文件路径
        rotation: 日志轮转大小
        retention: 日志保留时间
        json_format: 是否使用 JSON 格式
    """
    # 移除默认处理器
    logger.remove()
    
    # 控制台格式
    console_format = (
        "<green>{time:YYYY-MM-DD HH:mm:ss}</green> | "
        "<level>{level: <8}</level> | "
        "<cyan>{extra[request_id]}</cyan> | "
        "<cyan>{name}</cyan>:<cyan>{function}</cyan>:<cyan>{line}</cyan> - "
        "<level>{message}</level>"
    )
    
    # 文件格式
    file_format = (
        "{time:YYYY-MM-DD HH:mm:ss} | "
        "{level: <8} | "
        "{extra[request_id]} | "
        "{name}:{function}:{line} - "
        "{message}"
    )
    
    # JSON 格式
    json_format_str = (
        '{{"time": "{time:YYYY-MM-DD HH:mm:ss}", '
        '"level": "{level}", '
        '"request_id": "{extra[request_id]}", '
        '"logger": "{name}", '
        '"function": "{function}", '
        '"line": {line}, '
        '"message": "{message}"}}'
    )
    
    # 添加控制台处理器
    logger.add(
        sys.stdout,
        format=console_format,
        level=log_level,
        colorize=True,
        filter=lambda record: record["level"].name >= log_level
    )
    
    # 添加文件处理器
    if log_file:
        # 确保日志目录存在
        log_path = Path(log_file)
        log_path.parent.mkdir(parents=True, exist_ok=True)
        
        # 普通日志文件
        logger.add(
            log_file,
            format=json_format_str if json_format else file_format,
            level=log_level,
            rotation=rotation,
            retention=retention,
            compression="zip",
            encoding="utf-8",
            filter=lambda record: record["level"].name >= log_level
        )
        
        # 错误日志单独记录
        error_log_file = log_file.replace(".log", "_error.log")
        logger.add(
            error_log_file,
            format=json_format_str if json_format else file_format,
            level="ERROR",
            rotation=rotation,
            retention="30 days",
            compression="zip",
            encoding="utf-8"
        )
    
    # 设置默认 request_id
    logger.configure(extra={"request_id": "-"})


# 初始化日志
def init_logging():
    """初始化日志配置"""
    from config import get_settings
    
    settings = get_settings()
    
    setup_logging(
        log_level=settings.LOG.LEVEL,
        log_file=settings.LOG.FILE,
        rotation=settings.LOG.ROTATION,
        retention=settings.LOG.RETENTION
    )
    
    logger.info("日志系统初始化完成")
```

### 3.2 创建日志中间件

```python
"""
日志中间件
记录每个请求的详细信息
"""
import uuid
import time
from fastapi import Request, Response
from starlette.middleware.base import BaseHTTPMiddleware
from loguru import logger


class LoggingMiddleware(BaseHTTPMiddleware):
    """请求日志中间件"""
    
    async def dispatch(self, request: Request, call_next):
        # 生成请求 ID
        request_id = str(uuid.uuid4())[:8]
        
        # 绑定 request_id 到日志
        with logger.contextualize(request_id=request_id):
            # 记录请求
            logger.info(
                f"请求开始: {request.method} {request.url.path} "
                f"Client: {request.client.host if request.client else 'Unknown'}"
            )
            
            # 记录请求体（可选）
            # if request.method in ["POST", "PUT", "PATCH"]:
            #     body = await request.body()
            #     logger.debug(f"请求体: {body.decode('utf-8')}")
            
            # 记录开始时间
            start_time = time.time()
            
            try:
                # 调用下一个处理函数
                response: Response = await call_next(request)
                
                # 计算处理时间
                process_time = time.time() - start_time
                
                # 记录响应
                logger.info(
                    f"请求完成: {request.method} {request.url.path} "
                    f"状态码: {response.status_code} "
                    f"耗时: {process_time:.3f}s"
                )
                
                # 添加响应头
                response.headers["X-Request-ID"] = request_id
                response.headers["X-Process-Time"] = f"{process_time:.3f}s"
                
                # 慢请求警告
                if process_time > 1.0:
                    logger.warning(
                        f"慢请求警告: {request.method} {request.url.path} "
                        f"耗时: {process_time:.3f}s"
                    )
                
                return response
                
            except Exception as e:
                # 记录异常
                process_time = time.time() - start_time
                logger.exception(
                    f"请求异常: {request.method} {request.url.path} "
                    f"耗时: {process_time:.3f}s "
                    f"错误: {str(e)}"
                )
                raise
```

### 3.3 在主程序中使用

```python
"""
主程序 - 集成日志系统
"""
from fastapi import FastAPI, Request
from fastapi.middleware.cors import CORSMiddleware
from loguru import logger

# 初始化日志
from logging_config import init_logging
init_logging()

# 创建应用
app = FastAPI(title="日志系统示例")

# 添加日志中间件
from logging_middleware import LoggingMiddleware
app.add_middleware(LoggingMiddleware)

# CORS
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)


@app.get("/")
def root():
    """根路径"""
    logger.info("访问根路径")
    return {"message": "Hello World"}


@app.get("/error")
def trigger_error():
    """触发错误"""
    logger.warning("即将触发错误")
    raise Exception("这是一个测试错误")


@app.get("/slow")
def slow_api():
    """慢接口"""
    import time
    logger.info("开始处理慢接口")
    time.sleep(1.5)
    logger.info("慢接口处理完成")
    return {"message": "慢接口响应"}
```

## 四、日志级别

### 4.1 日志级别说明

| 级别 | 数值 | 说明 | 使用场景 |
|------|------|------|---------|
| TRACE | 5 | 追踪 | 详细的程序执行流程 |
| DEBUG | 10 | 调试 | 调试信息，开发环境使用 |
| INFO | 20 | 信息 | 正常的业务信息 |
| SUCCESS | 25 | 成功 | 操作成功信息 |
| WARNING | 30 | 警告 | 警告信息，不影响运行 |
| ERROR | 40 | 错误 | 错误信息，需要处理 |
| CRITICAL | 50 | 严重 | 严重错误，程序可能崩溃 |

### 4.2 不同环境的日志级别

```python
# 开发环境：DEBUG
LOG_LEVEL=DEBUG

# 测试环境：INFO
LOG_LEVEL=INFO

# 生产环境：WARNING
```

### 7.2 完整日志配置

```python
"""
日志配置 - 完整示例
"""
import sys
import json
from pathlib import Path
from loguru import logger
from typing import Optional


class LoggingConfig:
    """日志配置类"""
    
    def __init__(
        self,
        log_level: str = "INFO",
        log_dir: str = "logs",
        app_name: str = "app",
        rotation: str = "10 MB",
        retention: str = "7 days",
        json_format: bool = False,
        enable_console: bool = True,
        enable_file: bool = True
    ):
        self.log_level = log_level
        self.log_dir = Path(log_dir)
        self.app_name = app_name
        self.rotation = rotation
        self.retention = retention
        self.json_format = json_format
        self.enable_console = enable_console
        self.enable_file = enable_file
    
    def setup(self):
        """配置日志系统"""
        # 移除默认处理器
        logger.remove()
        
        # 确保日志目录存在
        if self.enable_file:
            self.log_dir.mkdir(parents=True, exist_ok=True)
        
        # 格式定义
        console_format = (
            "<green>{time:YYYY-MM-DD HH:mm:ss.SSS}</green> | "
            "<level>{level: <8}</level> | "
            "<cyan>{extra[request_id]: <8}</cyan> | "
            "<cyan>{name}</cyan>:<cyan>{function}</cyan>:<cyan>{line}</cyan> - "
            "<level>{message}</level>"
        )
        
        file_format = (
            "{time:YYYY-MM-DD HH:mm:ss.SSS} | "
            "{level: <8} | "
            "{extra[request_id]: <8} | "
            "{name}:{function}:{line} - "
            "{message}"
        )
        
        # 控制台输出
        if self.enable_console:
            logger.add(
                sys.stdout,
                format=console_format,
                level=self.log_level,
                colorize=True
            )
        
        # 文件输出
        if self.enable_file:
            # 普通日志
            logger.add(
                self.log_dir / f"{self.app_name}.log",
                format=file_format,
                level=self.log_level,
                rotation=self.rotation,
                retention=self.retention,
                compression="zip",
                encoding="utf-8"
            )
            
            # 错误日志
            logger.add(
                self.log_dir / f"{self.app_name}_error.log",
                format=file_format,
                level="ERROR",
                rotation=self.rotation,
                retention="30 days",
                compression="zip",
                encoding="utf-8"
            )
            
            # JSON 格式日志（用于日志收集）
            if self.json_format:
                logger.add(
                    self.log_dir / f"{self.app_name}.json",
                    format=self._json_format,
                    level=self.log_level,
                    rotation=self.rotation,
                    retention=self.retention,
                    compression="zip",
                    encoding="utf-8",
                    serialize=False
                )
        
        # 设置默认 request_id
        logger.configure(extra={"request_id": "-"})
        
        logger.info(f"日志系统初始化完成，日志级别: {self.log_level}")
    
    @staticmethod
    def _json_format(record):
        """JSON 格式化"""
        log_data = {
            "timestamp": record["time"].isoformat(),
            "level": record["level"].name,
            "message": record["message"],
            "logger": record["name"],
            "function": record["function"],
            "line": record["line"],
            "request_id": record["extra"].get("request_id", "-"),
            "extra": record["extra"]
        }
        return json.dumps(log_data, ensure_ascii=False) + "\n"


def init_logging():
    """初始化日志"""
    from config import get_settings
    
    settings = get_settings()
    
    config = LoggingConfig(
        log_level=settings.LOG.LEVEL,
        log_dir=Path(settings.LOG.FILE).parent,
        app_name="app",
        rotation=settings.LOG.ROTATION,
        retention=settings.LOG.RETENTION,
        json_format=False
    )
    
    config.setup()


# 初始化
init_logging()
```

### 8.2 日志分析脚本

```python
"""
日志分析脚本
"""
from collections import Counter
from datetime import datetime


def analyze_log(log_file: str):
    """分析日志文件"""
    level_count = Counter()
    error_messages = []
    slow_requests = []
    
    with open(log_file, 'r', encoding='utf-8') as f:
        for line in f:
            # 统计日志级别
            if 'ERROR' in line:
                level_count['ERROR'] += 1
                error_messages.append(line.strip())
            elif 'WARNING' in line:
                level_count['WARNING'] += 1
            elif 'INFO' in line:
                level_count['INFO'] += 1
            
            # 统计慢请求
            if '慢请求' in line:
                slow_requests.append(line.strip())
    
    print(f"日志级别统计: {dict(level_count)}")
    print(f"错误数量: {len(error_messages)}")
    print(f"慢请求数量: {len(slow_requests)}")
    
    # 打印前 5 个错误
    print("\n最近的错误:")
    for msg in error_messages[:5]:
        print(f"  {msg}")


if __name__ == "__main__":
    analyze_log("logs/app.log")
```
