## 一、异常处理简介

### 1.1 FastAPI 异常处理机制

FastAPI 提供了完善的异常处理机制：
- 自动处理 HTTP 异常
- 自动处理参数校验异常
- 支持自定义异常处理器
- 支持全局异常捕获

### 1.2 统一错误格式的重要性

**不统一的错误格式**：
```json
// 接口1
{"error": "用户不存在"}

// 接口2
{"message": "参数错误", "code": 400}

// 接口3
{"detail": "服务器错误"}
```

**统一的错误格式**：
```json
{
  "code": 400,
  "message": "参数错误",
  "data": null,
  "timestamp": "2024-01-15T10:30:00"
}
```

## 二、HTTPException

### 2.1 基本用法

```python
from fastapi import FastAPI, HTTPException

app = FastAPI()


@app.get("/users/{user_id}")
def get_user(user_id: int):
    """查询用户"""
    if user_id <= 0:
        raise HTTPException(
            status_code=400,
            detail="用户ID必须大于0"
        )
    
    if user_id > 100:
        raise HTTPException(
            status_code=404,
            detail="用户不存在"
        )
    
    return {"id": user_id, "name": f"用户{user_id}"}
```

### 2.2 自定义响应头

```python
@app.get("/auth")
def auth_check():
    """认证检查"""
    raise HTTPException(
        status_code=401,
        detail="未认证",
        headers={"WWW-Authenticate": "Bearer"}
    )
```

### 2.3 HTTPException 响应格式

默认格式：
```json
{
  "detail": "用户不存在"
}
```

## 三、自定义异常

### 3.1 定义业务异常

```python
from fastapi import FastAPI, HTTPException, Request
from fastapi.responses import JSONResponse
from pydantic import BaseModel
from typing import Optional
from datetime import datetime

app = FastAPI()


# 统一响应模型
class ErrorResponse(BaseModel):
    """统一错误响应模型"""
    code: int
    message: str
    data: Optional[dict] = None
    timestamp: str = Field(default_factory=lambda: datetime.now().isoformat())


# 自定义业务异常
class BusinessException(Exception):
    """业务异常基类"""
    def __init__(self, code: int, message: str, data: dict = None):
        self.code = code
        self.message = message
        self.data = data


class UserNotFoundException(BusinessException):
    """用户不存在异常"""
    def __init__(self, user_id: int):
        super().__init__(
            code=404,
            message="用户不存在",
            data={"user_id": user_id}
        )


class UserAlreadyExistsException(BusinessException):
    """用户已存在异常"""
    def __init__(self, username: str):
        super().__init__(
            code=400,
            message="用户名已存在",
            data={"username": username}
        )


class InsufficientStockException(BusinessException):
    """库存不足异常"""
    def __init__(self, product_id: int, required: int, available: int):
        super().__init__(
            code=400,
            message="库存不足",
            data={
                "product_id": product_id,
                "required": required,
                "available": available
            }
        )
```

### 3.2 注册异常处理器

```python
# 异常处理器
@app.exception_handler(BusinessException)
async def business_exception_handler(request: Request, exc: BusinessException):
    """业务异常处理器"""
    return JSONResponse(
        status_code=exc.code if exc.code < 500 else 400,
        content=ErrorResponse(
            code=exc.code,
            message=exc.message,
            data=exc.data
        ).model_dump()
    )


# 使用自定义异常
@app.get("/users/{user_id}")
def get_user(user_id: int):
    """查询用户"""
    if user_id > 100:
        raise UserNotFoundException(user_id)
    
    return {"id": user_id, "name": f"用户{user_id}"}


@app.post("/users")
def create_user(username: str):
    """创建用户"""
    if username == "admin":
        raise UserAlreadyExistsException(username)
    
    return {"id": 1, "username": username}


@app.post("/orders")
def create_order(product_id: int, quantity: int):
    """创建订单"""
    available_stock = 5
    
    if quantity > available_stock:
        raise InsufficientStockException(product_id, quantity, available_stock)
    
    return {"message": "订单创建成功"}
```

## 四、全局异常处理

### 4.1 处理 HTTPException

```python
from fastapi import FastAPI, HTTPException, Request
from fastapi.responses import JSONResponse
from fastapi.exceptions import HTTPException as FastAPIHTTPException

app = FastAPI()


@app.exception_handler(FastAPIHTTPException)
async def http_exception_handler(request: Request, exc: FastAPIHTTPException):
    """HTTP 异常处理器"""
    return JSONResponse(
        status_code=exc.status_code,
        content={
            "code": exc.status_code,
            "message": exc.detail,
            "data": None,
            "timestamp": datetime.now().isoformat()
        }
    )
```

### 4.2 处理参数校验异常

```python
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
from fastapi.exceptions import RequestValidationError

app = FastAPI()


@app.exception_handler(RequestValidationError)
async def validation_exception_handler(request: Request, exc: RequestValidationError):
    """参数校验异常处理器"""
    # 提取错误信息
    errors = []
    for error in exc.errors():
        field = " -> ".join(str(loc) for loc in error["loc"])
        message = error["msg"]
        errors.append({
            "field": field,
            "message": message,
            "type": error["type"]
        })
    
    return JSONResponse(
        status_code=422,
        content={
            "code": 422,
            "message": "参数校验失败",
            "data": {"errors": errors},
            "timestamp": datetime.now().isoformat()
        }
    )
```

### 4.3 处理所有异常

```python
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
import traceback

app = FastAPI()


@app.exception_handler(Exception)
async def global_exception_handler(request: Request, exc: Exception):
    """全局异常处理器"""
    # 记录错误日志
    error_trace = traceback.format_exc()
    print(f"[错误] {error_trace}")
    
    return JSONResponse(
        status_code=500,
        content={
            "code": 500,
            "message": "服务器内部错误",
            "data": {
                "error": str(exc),
                "trace": error_trace if app.debug else None
            },
            "timestamp": datetime.now().isoformat()
        }
    )
```

## 五、完整异常处理示例

```python
from fastapi import FastAPI, Request, HTTPException
from fastapi.responses import JSONResponse
from fastapi.exceptions import RequestValidationError
from pydantic import BaseModel, Field
from typing import Optional, Any
from datetime import datetime
import traceback


# ==================== 统一响应模型 ====================

class ResponseModel(BaseModel):
    """统一响应模型"""
    code: int = Field(default=200, description="状态码")
    message: str = Field(default="success", description="提示信息")
    data: Optional[Any] = Field(default=None, description="返回数据")
    timestamp: str = Field(default_factory=lambda: datetime.now().isoformat())


# ==================== 自定义异常 ====================

class AppException(Exception):
    """应用异常基类"""
    def __init__(self, code: int = 400, message: str = "业务错误", data: Any = None):
        self.code = code
        self.message = message
        self.data = data
        super().__init__(self.message)


class NotFoundException(AppException):
    """资源不存在异常"""
    def __init__(self, resource: str, resource_id: int):
        super().__init__(
            code=404,
            message=f"{resource}不存在",
            data={"resource": resource, "id": resource_id}
        )


class BadRequestException(AppException):
    """错误请求异常"""
    def __init__(self, message: str = "请求参数错误", data: Any = None):
        super().__init__(code=400, message=message, data=data)


class UnauthorizedException(AppException):
    """未认证异常"""
    def __init__(self, message: str = "未认证，请先登录"):
        super().__init__(code=401, message=message)


class ForbiddenException(AppException):
    """无权限异常"""
    def __init__(self, message: str = "无权限访问"):
        super().__init__(code=403, message=message)


class ConflictException(AppException):
    """资源冲突异常"""
    def __init__(self, message: str = "资源冲突", data: Any = None):
        super().__init__(code=409, message=message, data=data)


# ==================== FastAPI 应用 ====================

app = FastAPI(
    title="异常处理示例",
    description="演示统一异常处理",
    version="1.0.0"
)


# ==================== 异常处理器 ====================

@app.exception_handler(AppException)
async def app_exception_handler(request: Request, exc: AppException):
    """应用异常处理器"""
    return JSONResponse(
        status_code=exc.code if exc.code < 500 else 400,
        content=ResponseModel(
            code=exc.code,
            message=exc.message,
            data=exc.data
        ).model_dump()
    )


@app.exception_handler(RequestValidationError)
async def validation_exception_handler(request: Request, exc: RequestValidationError):
    """参数校验异常处理器"""
    errors = []
    for error in exc.errors():
        field = " -> ".join(str(loc) for loc in error["loc"])
        errors.append({
            "field": field,
            "message": error["msg"],
            "type": error["type"]
        })
    
    return JSONResponse(
        status_code=422,
        content=ResponseModel(
            code=422,
            message="参数校验失败",
            data={"errors": errors}
        ).model_dump()
    )


@app.exception_handler(HTTPException)
async def http_exception_handler(request: Request, exc: HTTPException):
    """HTTP 异常处理器"""
    return JSONResponse(
        status_code=exc.status_code,
        content=ResponseModel(
            code=exc.status_code,
            message=exc.detail,
            data=None
        ).model_dump()
    )


@app.exception_handler(Exception)
async def global_exception_handler(request: Request, exc: Exception):
    """全局异常处理器"""
    error_trace = traceback.format_exc()
    print(f"[错误] {error_trace}")
    
    return JSONResponse(
        status_code=500,
        content=ResponseModel(
            code=500,
            message="服务器内部错误",
            data={"error": str(exc)} if app.debug else None
        ).model_dump()
    )


# ==================== 接口示例 ====================

# 模拟数据库
fake_db = {
    1: {"id": 1, "name": "商品1", "price": 99.9, "stock": 10},
    2: {"id": 2, "name": "商品2", "price": 199.9, "stock": 5},
}


class ProductCreate(BaseModel):
    """创建商品请求模型"""
    name: str = Field(..., min_length=1, max_length=100)
    price: float = Field(..., gt=0)
    stock: int = Field(default=0, ge=0)


class OrderCreate(BaseModel):
    """创建订单请求模型"""
    product_id: int
    quantity: int = Field(..., gt=0)


@app.get("/", response_model=ResponseModel)
def root():
    """根路径"""
    return ResponseModel(message="服务运行正常")


@app.get("/products/{product_id}", response_model=ResponseModel)
def get_product(product_id: int):
    """查询商品"""
    product = fake_db.get(product_id)
    
    if not product:
        raise NotFoundException("商品", product_id)
    
    return ResponseModel(data=product)


@app.post("/products", response_model=ResponseModel)
def create_product(product: ProductCreate):
    """创建商品"""
    # 检查商品名是否已存在
    for p in fake_db.values():
        if p["name"] == product.name:
            raise ConflictException(
                message="商品名已存在",
                data={"name": product.name}
            )
    
    # 创建商品
    new_id = max(fake_db.keys()) + 1
    fake_db[new_id] = {
        "id": new_id,
        "name": product.name,
        "price": product.price,
        "stock": product.stock
    }
    
    return ResponseModel(message="商品创建成功", data=fake_db[new_id])


@app.post("/orders", response_model=ResponseModel)
def create_order(order: OrderCreate):
    """创建订单"""
    # 检查商品是否存在
    product = fake_db.get(order.product_id)
    if not product:
        raise NotFoundException("商品", order.product_id)
    
    # 检查库存是否充足
    if order.quantity > product["stock"]:
        raise BadRequestException(
            message="库存不足",
            data={
                "product_id": order.product_id,
                "required": order.quantity,
                "available": product["stock"]
            }
        )
    
    # 扣减库存
    product["stock"] -= order.quantity
    
    return ResponseModel(
        message="订单创建成功",
        data={
            "product_id": order.product_id,
            "quantity": order.quantity,
            "remaining_stock": product["stock"]
        }
    )


@app.get("/error")
def trigger_error():
    """触发错误（测试全局异常处理）"""
    raise Exception("这是一个测试错误")


@app.get("/auth")
def check_auth():
    """检查认证"""
    raise UnauthorizedException("请先登录")


@app.get("/admin")
def check_admin():
    """检查管理员权限"""
    raise ForbiddenException("需要管理员权限")
```

## 八、常见问题

### Q1: 如何区分业务异常和系统异常？

- **业务异常**：用户操作导致的错误，如参数错误、资源不存在
- **系统异常**：程序 bug 或外部服务故障

```python
# 业务异常：返回给用户
if not user:
    raise NotFoundException("用户", user_id)

# 系统异常：记录日志，返回友好提示
try:
    result = external_api.call()
except Exception as e:
    logger.error(f"外部服务调用失败: {e}")
    raise AppException(code=500, message="服务暂时不可用")
```

### Q2: 如何记录异常日志？

```python
import logging

logger = logging.getLogger(__name__)


@app.exception_handler(Exception)
async def global_exception_handler(request: Request, exc: Exception):
    # 记录详细日志
    logger.error(
        f"异常: {exc}",
        extra={
            "request": str(request.url),
            "method": request.method,
            "traceback": traceback.format_exc()
        }
    )
    
    # 返回简化响应
    return JSONResponse(...)
```

### Q3: 如何在开发环境显示详细错误？

```python
from fastapi import FastAPI

app = FastAPI(debug=True)  # 开发环境开启 debug


@app.exception_handler(Exception)

    return JSONResponse(
        status_code=500,
        content={
            "code": 500,
            "message": str(exc),
            "trace": traceback.format_exc() if app.debug else None
        }
    )
```
