## 二、文件上传

### 2.1 单文件上传

```python
"""
文件上传接口
"""
from fastapi import FastAPI, File, UploadFile
from pathlib import Path
import shutil
from datetime import datetime

app = FastAPI()

# 上传目录
upload_dir = Path("static/uploads")
upload_dir.mkdir(parents=True, exist_ok=True)


@app.post("/upload")
async def upload_file(file: UploadFile = File(...)):
    """
    单文件上传
    
    - file: 上传的文件对象
    """
    # 生成唯一文件名
    timestamp = datetime.now().strftime("%Y%m%d%H%M%S")
    file_ext = Path(file.filename).suffix
    new_filename = f"{timestamp}_{file.filename}"
    
    # 保存文件
    file_path = upload_dir / new_filename
    with open(file_path, "wb") as buffer:
        shutil.copyfileobj(file.file, buffer)
    
    return {
        "filename": new_filename,
        "original_name": file.filename,
        "content_type": file.content_type,
        "size": file_path.stat().st_size,
        "url": f"/static/uploads/{new_filename}"
    }
```

### 2.2 多文件上传

```python
from typing import List

@app.post("/upload/multiple")
async def upload_multiple_files(files: List[UploadFile] = File(...)):
    """
    多文件上传
    
    - files: 上传的文件列表
    """
    uploaded_files = []
    
    for file in files:
        # 生成唯一文件名
        timestamp = datetime.now().strftime("%Y%m%d%H%M%S%f")
        file_ext = Path(file.filename).suffix
        new_filename = f"{timestamp}_{file.filename}"
        
        # 保存文件
        file_path = upload_dir / new_filename
        with open(file_path, "wb") as buffer:
            shutil.copyfileobj(file.file, buffer)
        
        uploaded_files.append({
            "filename": new_filename,
            "original_name": file.filename,
            "size": file_path.stat().st_size,
            "url": f"/static/uploads/{new_filename}"
        })
    
    return {
        "count": len(uploaded_files),
        "files": uploaded_files
    }
```

### 2.3 文件大小限制

```python
from fastapi import FastAPI, File, UploadFile, HTTPException

app = FastAPI()

# 最大文件大小（字节）
MAX_FILE_SIZE = 10 * 1024 * 1024  # 10MB


@app.post("/upload/limited")
async def upload_with_limit(file: UploadFile = File(...)):
    """
    带大小限制的文件上传
    """
    # 读取文件内容
    contents = await file.read()
    file_size = len(contents)
    
    # 检查文件大小
    if file_size > MAX_FILE_SIZE:
        raise HTTPException(
            status_code=413,
            detail=f"文件过大，最大允许 {MAX_FILE_SIZE / 1024 / 1024}MB"
        )
    
    # 保存文件
    file_path = upload_dir / file.filename
    with open(file_path, "wb") as buffer:
        buffer.write(contents)
    
    return {
        "filename": file.filename,
        "size": file_size
    }
```

### 2.4 文件类型限制

```python
from fastapi import FastAPI, File, UploadFile, HTTPException
from pathlib import Path

app = FastAPI()

# 允许的文件类型
ALLOWED_EXTENSIONS = {".jpg", ".jpeg", ".png", ".gif", ".pdf", ".doc", ".docx"}


@app.post("/upload/filtered")
async def upload_with_filter(file: UploadFile = File(...)):
    """
    带类型限制的文件上传
    """
    # 检查文件扩展名
    file_ext = Path(file.filename).suffix.lower()
    if file_ext not in ALLOWED_EXTENSIONS:
        raise HTTPException(
            status_code=400,
            detail=f"不支持的文件类型: {file_ext}。允许的类型: {ALLOWED_EXTENSIONS}"
        )
    
    # 保存文件
    file_path = upload_dir / file.filename
    with open(file_path, "wb") as buffer:
        shutil.copyfileobj(file.file, buffer)
    
    return {
        "filename": file.filename,
        "type": file_ext
    }
```

### 3.2 配置模板

```python
"""
模板渲染配置
"""
from fastapi import FastAPI, Request
from fastapi.staticfiles import StaticFiles
from fastapi.templating import Jinja2Templates
from pathlib import Path

app = FastAPI()

# 静态文件
app.mount("/static", StaticFiles(directory="static"), name="static")

# 模板配置
templates = Jinja2Templates(directory="templates")


@app.get("/page")
def page(request: Request):
    """渲染模板页面"""
    return templates.TemplateResponse(
        "index.html",
        {
            "request": request,
            "title": "首页",
            "message": "Hello FastAPI"
        }
    )
```

## 四、完整示例

### 4.1 主程序

```python
"""
主程序 - 静态文件与模板
"""
from fastapi import FastAPI, Request, File, UploadFile, HTTPException
from fastapi.staticfiles import StaticFiles
from fastapi.templating import Jinja2Templates
from fastapi.responses import HTMLResponse
from pathlib import Path
import shutil
from datetime import datetime
from typing import List

app = FastAPI(title="静态文件与模板示例")

# 目录配置
static_dir = Path("static")
upload_dir = static_dir / "uploads"
templates_dir = Path("templates")

# 创建目录
static_dir.mkdir(exist_ok=True)
upload_dir.mkdir(parents=True, exist_ok=True)
templates_dir.mkdir(exist_ok=True)

# 挂载静态文件
app.mount("/static", StaticFiles(directory="static"), name="static")

# 模板配置
templates = Jinja2Templates(directory="templates")


# ==================== 页面路由 ====================

@app.get("/", response_class=HTMLResponse)
def index(request: Request):
    """首页"""
    return templates.TemplateResponse(
        "index.html",
        {
            "request": request,
            "title": "FastAPI CRUD Tutorial",
            "message": "欢迎使用 FastAPI CRUD 实战教程"
        }
    )


@app.get("/users", response_class=HTMLResponse)
def users_list(request: Request, page: int = 1):
    """用户列表页面"""
    # 模拟数据
    users = [
        {"id": 1, "username": "user1", "email": "user1@example.com", "status": "active", "created_at": "2024-01-15"},
        {"id": 2, "username": "user2", "email": "user2@example.com", "status": "inactive", "created_at": "2024-01-16"},
        {"id": 3, "username": "user3", "email": "user3@example.com", "status": "active", "created_at": "2024-01-17"},
    ]
    
    return templates.TemplateResponse(
        "users/list.html",
        {
            "request": request,
            "users": users,
            "page": page,
            "has_next": False
        }
    )


@app.get("/upload", response_class=HTMLResponse)
def upload_page(request: Request):
    """文件上传页面"""
    return templates.TemplateResponse(
        "upload.html",
        {"request": request}
    )


# ==================== 文件上传接口 ====================

@app.post("/api/upload")
async def upload_file(file: UploadFile = File(...)):
    """单文件上传"""
    # 生成文件名
    timestamp = datetime.now().strftime("%Y%m%d%H%M%S%f")
    file_ext = Path(file.filename).suffix
    new_filename = f"{timestamp}{file_ext}"
    
    # 保存文件
    file_path = upload_dir / new_filename
    with open(file_path, "wb") as buffer:
        shutil.copyfileobj(file.file, buffer)
    
    return {
        "code": 200,
        "message": "上传成功",
        "data": {
            "filename": new_filename,
            "original_name": file.filename,
            "size": file_path.stat().st_size,
            "url": f"/static/uploads/{new_filename}"
        }
    }


@app.post("/api/upload/multiple")
async def upload_multiple(files: List[UploadFile] = File(...)):
    """多文件上传"""
    uploaded = []
    
    for file in files:
        timestamp = datetime.now().strftime("%Y%m%d%H%M%S%f")
        file_ext = Path(file.filename).suffix
        new_filename = f"{timestamp}{file_ext}"
        
        file_path = upload_dir / new_filename
        with open(file_path, "wb") as buffer:
            shutil.copyfileobj(file.file, buffer)
        
        uploaded.append({
            "filename": new_filename,
            "original_name": file.filename,
            "size": file_path.stat().st_size
        })
    
    return {
        "code": 200,
        "message": f"成功上传 {len(uploaded)} 个文件",
        "data": uploaded
    }
```
