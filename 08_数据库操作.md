## 一、SQLAlchemy 简介

### 1.1 什么是 SQLAlchemy？

SQLAlchemy 是 Python 最流行的 ORM（对象关系映射）框架，提供了：
- ✅ SQL 表达式语言
- ✅ ORM 映射
- ✅ 数据库连接池
- ✅ 事务管理

### 1.2 ORM 的优势

**传统 SQL 方式**：
```python
import sqlite3

conn = sqlite3.connect('test.db')
cursor = conn.cursor()
cursor.execute("SELECT * FROM users WHERE id = ?", (1,))
user = cursor.fetchone()
conn.close()
```

**ORM 方式**：
```python
from sqlalchemy.orm import Session

db = SessionLocal()
user = db.query(User).filter(User.id == 1).first()
db.close()
```

## 二、数据库连接配置

### 2.1 创建 database.py

```python
"""
数据库连接配置
使用 SQLAlchemy 创建 SQLite 数据库连接
"""
from sqlalchemy import create_engine
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker
import os

# 数据库文件路径
# SQLite 会自动创建文件，无需额外安装
SQLALCHEMY_DATABASE_URL = "sqlite:///./test_api.db"

# 创建数据库引擎
# connect_args={"check_same_thread": False} 仅用于 SQLite
# 允许多线程访问同一个数据库连接
engine = create_engine(
    SQLALCHEMY_DATABASE_URL,
    connect_args={"check_same_thread": False},
    echo=False  # True 会打印 SQL 语句，调试时开启
)

# 创建会话工厂
# autocommit=False: 不自动提交，需要手动 commit
# autoflush=False: 不自动刷新到数据库
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)

# 创建基类
# 所有模型类都将继承这个基类
Base = declarative_base()


# 依赖函数：获取数据库会话
def get_db():
    """
    获取数据库会话
    
    使用 yield 确保请求结束后自动关闭会话
    这是 FastAPI 依赖注入的标准模式
    """
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()
```

## 三、ORM 模型定义

### 3.1 创建 models.py

```python
"""
SQLAlchemy ORM 模型定义
定义数据库表结构
"""
from sqlalchemy import Column, Integer, String, DateTime, Text, Boolean, Float
from sqlalchemy.sql import func
from sqlalchemy import ForeignKey
from sqlalchemy.orm import relationship
from database import Base


class User(Base):
    """
    用户表模型
    
    __tablename__: 数据库表名
    Column: 定义字段
    """
    __tablename__ = "users"
    
    # 主键：自增 ID
    id = Column(Integer, primary_key=True, index=True, autoincrement=True)
    
    # 用户名：唯一、不能为空、创建索引
    username = Column(String(50), unique=True, nullable=False, index=True)
    
    # 邮箱：唯一、不能为空
    email = Column(String(100), unique=True, nullable=False)
    
    # 密码：不能为空
    password = Column(String(255), nullable=False)
    
    # 手机号：可为空
    phone = Column(String(20), nullable=True)
    
    # 用户状态：active/inactive
    status = Column(String(20), default="active")
    
    # 是否是管理员
    is_admin = Column(Boolean, default=False)
    
    # 备注信息
    remark = Column(Text, nullable=True)
    
    # 创建时间：自动记录插入时间
    created_at = Column(DateTime(timezone=True), server_default=func.now())
    
    # 更新时间：自动记录更新时间
    updated_at = Column(DateTime(timezone=True), onupdate=func.now())
    
    # 关系：一个用户有多个订单
    orders = relationship("Order", back_populates="user")


class Product(Base):
    """
    商品表模型
    """
    __tablename__ = "products"
    
    id = Column(Integer, primary_key=True, index=True, autoincrement=True)
    
    # 商品名称
    name = Column(String(100), nullable=False, index=True)
    
    # 商品描述
    description = Column(Text, nullable=True)
    
    # 价格：单位分（整数存储，避免浮点数精度问题）
    price = Column(Integer, nullable=False)
    
    # 库存数量
    stock = Column(Integer, default=0)
    
    # 商品状态：on_sale/off_sale
    status = Column(String(20), default="on_sale")
    
    # 分类 ID（外键）
    category_id = Column(Integer, ForeignKey("categories.id"), nullable=True)
    
    created_at = Column(DateTime(timezone=True), server_default=func.now())
    updated_at = Column(DateTime(timezone=True), onupdate=func.now())
    
    # 关系
    category = relationship("Category", back_populates="products")
    order_items = relationship("OrderItem", back_populates="product")


class Category(Base):
    """
    商品分类表模型
    """
    __tablename__ = "categories"
    
    id = Column(Integer, primary_key=True, index=True)
    name = Column(String(50), nullable=False, unique=True)
    description = Column(Text, nullable=True)
    
    created_at = Column(DateTime(timezone=True), server_default=func.now())
    
    # 关系：一个分类有多个商品
    products = relationship("Product", back_populates="category")


class Order(Base):
    """
    订单表模型
    """
    __tablename__ = "orders"
    
    id = Column(Integer, primary_key=True, index=True)
    
    # 订单号
    order_no = Column(String(50), unique=True, nullable=False, index=True)
    
    # 用户 ID（外键）
    user_id = Column(Integer, ForeignKey("users.id"), nullable=False)
    
    # 订单状态：pending/paid/shipped/completed/cancelled
    status = Column(String(20), default="pending")
    
    # 订单总金额（分）
    total_amount = Column(Integer, default=0)
    
    # 收货地址
    shipping_address = Column(Text, nullable=True)
    
    # 备注
    remark = Column(Text, nullable=True)
    
    created_at = Column(DateTime(timezone=True), server_default=func.now())
    updated_at = Column(DateTime(timezone=True), onupdate=func.now())
    
    # 关系
    user = relationship("User", back_populates="orders")
    order_items = relationship("OrderItem", back_populates="order")


class OrderItem(Base):
    """
    订单明细表模型
    """
    __tablename__ = "order_items"
    
    id = Column(Integer, primary_key=True, index=True)
    
    # 订单 ID（外键）
    order_id = Column(Integer, ForeignKey("orders.id"), nullable=False)
    
    # 商品 ID（外键）
    product_id = Column(Integer, ForeignKey("products.id"), nullable=False)
    
    # 商品数量
    quantity = Column(Integer, nullable=False)
    
    # 商品单价（分）
    price = Column(Integer, nullable=False)
    
    # 小计金额（分）
    subtotal = Column(Integer, nullable=False)
    
    created_at = Column(DateTime(timezone=True), server_default=func.now())
    
    # 关系
    order = relationship("Order", back_populates="order_items")
    product = relationship("Product", back_populates="order_items")
```

```python
"""
Pydantic 数据校验模型
用于请求参数校验和响应数据格式化
"""
from pydantic import BaseModel, Field, EmailStr, field_validator
from typing import Optional, List
from datetime import datetime


# ==================== 统一响应格式 ====================

class ResponseModel(BaseModel):
    """统一响应格式"""
    code: int = Field(default=200, description="状态码")
    message: str = Field(default="success", description="提示信息")
    data: Optional[dict | list] = Field(default=None, description="返回数据")
    timestamp: str = Field(default_factory=lambda: datetime.now().isoformat())


# ==================== 用户相关模型 ====================

class UserCreate(BaseModel):
    """创建用户的请求模型"""
    username: str = Field(..., min_length=2, max_length=50, description="用户名")
    email: EmailStr = Field(..., description="邮箱")
    password: str = Field(..., min_length=6, max_length=50, description="密码")
    phone: Optional[str] = Field(None, max_length=20, description="手机号")
    status: str = Field(default="active", description="状态")
    is_admin: bool = Field(default=False, description="是否管理员")
    remark: Optional[str] = Field(None, description="备注")
    
    @field_validator('status')
    @classmethod
    def validate_status(cls, v):
        if v not in ['active', 'inactive']:
            raise ValueError('status 必须是 active 或 inactive')
        return v
    
    class Config:
        json_schema_extra = {
            "example": {
                "username": "testuser",
                "email": "test@example.com",
                "password": "password123",
                "phone": "13800138000",
                "status": "active",
                "is_admin": False
            }
        }


class UserUpdate(BaseModel):
    """更新用户的请求模型"""
    username: Optional[str] = Field(None, min_length=2, max_length=50)
    email: Optional[EmailStr] = None
    phone: Optional[str] = Field(None, max_length=20)
    status: Optional[str] = None
    remark: Optional[str] = None
    
    @field_validator('status')
    @classmethod
    def validate_status(cls, v):
        if v is not None and v not in ['active', 'inactive']:
            raise ValueError('status 必须是 active 或 inactive')
        return v


class UserResponse(BaseModel):
    """用户响应模型"""
    id: int
    username: str
    email: str
    phone: Optional[str]
    status: str
    is_admin: bool
    remark: Optional[str]
    created_at: Optional[datetime]
    updated_at: Optional[datetime]
    
    class Config:
        from_attributes = True  # 兼容 SQLAlchemy ORM 对象


# ==================== 商品相关模型 ====================

    name: str = Field(..., min_length=1, max_length=100)
    description: Optional[str] = None

    status: str = Field(default="on_sale")

class ProductUpdate(BaseModel):
```

```python
"""
CRUD 数据库操作函数
封装所有数据库增删改查操作
"""
from sqlalchemy.orm import Session
from sqlalchemy import or_, and_
from typing import Optional, List
import models
import schemas
import uuid
from datetime import datetime


# ==================== 用户 CRUD ====================

def create_user(db: Session, user: schemas.UserCreate):
    """创建用户"""
    db_user = models.User(
        username=user.username,
        email=user.email,
        password=user.password,  # 实际项目应该加密
        phone=user.phone,
        status=user.status,
        is_admin=user.is_admin,
        remark=user.remark
    )
    db.add(db_user)
    db.commit()
    db.refresh(db_user)
    return db_user


def get_user_by_id(db: Session, user_id: int):
    """根据 ID 查询用户"""
    return db.query(models.User).filter(models.User.id == user_id).first()


def get_user_by_username(db: Session, username: str):
    """根据用户名查询用户"""
    return db.query(models.User).filter(models.User.username == username).first()


def get_user_by_email(db: Session, email: str):
    """根据邮箱查询用户"""
    return db.query(models.User).filter(models.User.email == email).first()


def get_users(
    db: Session,
    skip: int = 0,
    limit: int = 100,
    status: Optional[str] = None,
    keyword: Optional[str] = None
):
    """查询用户列表"""
    query = db.query(models.User)
    
    if status:
        query = query.filter(models.User.status == status)
    
    if keyword:
        query = query.filter(
            or_(
                models.User.username.like(f"%{keyword}%"),
                models.User.email.like(f"%{keyword}%")
            )
        )
    
    query = query.order_by(models.User.created_at.desc())
    return query.offset(skip).limit(limit).all()


def update_user(db: Session, user_id: int, user: schemas.UserUpdate):
    """更新用户"""
    db_user = get_user_by_id(db, user_id)
    if not db_user:
        return None
    
    update_data = user.model_dump(exclude_unset=True)
    for key, value in update_data.items():
        setattr(db_user, key, value)
    
    db.commit()
    db.refresh(db_user)
    return db_user


def delete_user(db: Session, user_id: int):
    """删除用户"""
    db_user = get_user_by_id(db, user_id)
    if not db_user:
        return None
    
    db.delete(db_user)
    db.commit()
    return db_user

def get_products(
    db: Session,
    skip: int = 0,
    limit: int = 100,
    status: Optional[str] = None,
    keyword: Optional[str] = None,
    category_id: Optional[int] = None,
    min_price: Optional[int] = None,
    max_price: Optional[int] = None
):
    """查询商品列表"""
    query = db.query(models.Product)
    
    if status:
        query = query.filter(models.Product.status == status)
    
    if keyword:

    if category_id:
        query = query.filter(models.Product.category_id == category_id)
    
    if min_price is not None:
        query = query.filter(models.Product.price >= min_price)
    
    if max_price is not None:
        query = query.filter(models.Product.price <= max_price)
    
    query = query.order_by(models.Product.created_at.desc())
    return query.offset(skip).limit(limit).all()


def update_product(db: Session, product_id: int, product: schemas.ProductUpdate):

# ==================== 订单 CRUD ====================

def create_order(db: Session, user_id: int, order: schemas.OrderCreate):
    """创建订单"""
    # 生成订单号
    order_no = f"ORD{datetime.now().strftime('%Y%m%d%H%M%S')}{uuid.uuid4().hex[:6].upper()}"
    
    # 创建订单
    db_order = models.Order(
        order_no=order_no,
        user_id=user_id,
        shipping_address=order.shipping_address,
        remark=order.remark
    )
    db.add(db_order)
    db.flush()  # 获取订单 ID
    
    # 创建订单明细
    total_amount = 0
    for item in order.items:
        # 查询商品
        product = get_product_by_id(db, item.product_id)
        if not product:
            raise ValueError(f"商品 {item.product_id} 不存在")
        
        if product.stock < item.quantity:
            raise ValueError(f"商品 {product.name} 库存不足")
        
        # 计算小计
        subtotal = product.price * item.quantity
        total_amount += subtotal
        
        # 创建订单明细
        db_item = models.OrderItem(
            order_id=db_order.id,
            product_id=item.product_id,
            quantity=item.quantity,
            price=product.price,
            subtotal=subtotal
        )
        db.add(db_item)
        
        # 扣减库存
        product.stock -= item.quantity
    
    # 更新订单总金额
    db_order.total_amount = total_amount
    
    db.commit()
    db.refresh(db_order)
    return db_order


def get_order_by_id(db: Session, order_id: int):
    """根据 ID 查询订单"""
    return db.query(models.Order).filter(models.Order.id == order_id).first()


def get_orders(
    db: Session,
    user_id: Optional[int] = None,
    status: Optional[str] = None,
    skip: int = 0,
    limit: int = 100
):
    """查询订单列表"""
    query = db.query(models.Order)
    
    if user_id:
        query = query.filter(models.Order.user_id == user_id)
    
    if status:
        query = query.filter(models.Order.status == status)
    
    query = query.order_by(models.Order.created_at.desc())
    return query.offset(skip).limit(limit).all()


def update_order_status(db: Session, order_id: int, status: str):
    """更新订单状态"""
    db_order = get_order_by_id(db, order_id)
    if not db_order:
        return None
    
    db_order.status = status
    db.commit()
    db.refresh(db_order)
    return db_order
```

## 六、路由实现

### 6.1 创建 routers/users.py

```python
"""
用户路由
"""
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.orm import Session
from typing import Optional
from database import get_db
import crud
import schemas
from schemas import ResponseModel

router = APIRouter(prefix="/users", tags=["用户管理"])


@router.post("/", response_model=ResponseModel, summary="创建用户")
def create_user(user: schemas.UserCreate, db: Session = Depends(get_db)):
    """创建用户"""
    # 检查用户名是否已存在
    if crud.get_user_by_username(db, user.username):
        raise HTTPException(status_code=400, detail="用户名已存在")
    
    # 检查邮箱是否已存在
    if crud.get_user_by_email(db, user.email):
        raise HTTPException(status_code=400, detail="邮箱已被注册")
    
    # 创建用户
    new_user = crud.create_user(db, user)
    
    return ResponseModel(
        message="用户创建成功",
        data=schemas.UserResponse.model_validate(new_user).model_dump()
    )


@router.get("/{user_id}", response_model=ResponseModel, summary="查询单个用户")
def get_user(user_id: int, db: Session = Depends(get_db)):
    """查询单个用户"""
    user = crud.get_user_by_id(db, user_id)
    if not user:
        raise HTTPException(status_code=404, detail="用户不存在")
    
    return ResponseModel(
        data=schemas.UserResponse.model_validate(user).model_dump()
    )


@router.get("/", response_model=ResponseModel, summary="查询用户列表")
def get_users(
    skip: int = 0,
    limit: int = 10,
    status: Optional[str] = None,
    keyword: Optional[str] = None,
    db: Session = Depends(get_db)
):
    """查询用户列表"""
    users = crud.get_users(db, skip=skip, limit=limit, status=status, keyword=keyword)
    users_data = [schemas.UserResponse.model_validate(u).model_dump() for u in users]
    
    return ResponseModel(
        message=f"查询成功，共 {len(users_data)} 条记录",
        data=users_data
    )


@router.put("/{user_id}", response_model=ResponseModel, summary="更新用户")
def update_user(user_id: int, user: schemas.UserUpdate, db: Session = Depends(get_db)):
    """更新用户"""
    # 检查用户是否存在
    db_user = crud.get_user_by_id(db, user_id)
    if not db_user:
        raise HTTPException(status_code=404, detail="用户不存在")
    
    # 检查用户名是否重复
    if user.username:
        existing = crud.get_user_by_username(db, user.username)
        if existing and existing.id != user_id:
            raise HTTPException(status_code=400, detail="用户名已存在")
    
    # 检查邮箱是否重复
    if user.email:
        existing = crud.get_user_by_email(db, user.email)
        if existing and existing.id != user_id:
            raise HTTPException(status_code=400, detail="邮箱已被注册")
    
    # 更新用户
    updated_user = crud.update_user(db, user_id, user)
    
    return ResponseModel(
        message="用户更新成功",
        data=schemas.UserResponse.model_validate(updated_user).model_dump()
    )


@router.delete("/{user_id}", response_model=ResponseModel, summary="删除用户")
def delete_user(user_id: int, db: Session = Depends(get_db)):
    """删除用户"""
    user = crud.delete_user(db, user_id)
    if not user:
        raise HTTPException(status_code=404, detail="用户不存在")
    
    return ResponseModel(
        message="用户删除成功",
        data={"deleted_user_id": user_id}
    )
```
