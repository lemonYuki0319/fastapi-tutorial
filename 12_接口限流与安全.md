## 一、接口限流

### 1.1 为什么需要限流？

- ✅ 防止恶意刷接口
- ✅ 保护服务器资源
- ✅ 防止 DDoS 攻击
- ✅ 公平分配资源

### 1.2 使用 slowapi 实现限流

**安装依赖**：
```bash
pip install slowapi
```

**基本配置**：
```python
"""
接口限流配置
"""
from fastapi import FastAPI, Request
from slowapi import Limiter, _rate_limit_exceeded_handler
from slowapi.util import get_remote_address
from slowapi.errors import RateLimitExceeded

app = FastAPI()

# 创建限流器
limiter = Limiter(key_func=get_remote_address)
app.state.limiter = limiter
app.add_exception_handler(RateLimitExceeded, _rate_limit_exceeded_handler)


# 限流装饰器
@app.get("/")
@limiter.limit("10/minute")  # 每分钟 10 次
def root(request: Request):
    return {"message": "Hello World"}


# 多个限流规则
@app.get("/api/data")
@limiter.limit(["10/minute", "100/hour"])  # 每分钟 10 次且每小时 100 次
def get_data(request: Request):
    return {"data": []}
```

### 1.3 限流规则说明

```python
# 时间单位
"10/second"   # 每秒 10 次
"10/minute"   # 每分钟 10 次
"10/hour"     # 每小时 10 次
"10/day"      # 每天 10 次

# 组合规则
["10/minute", "100/hour", "1000/day"]

# 动态限流
def dynamic_limit(request: Request):
    # 根据用户等级返回不同的限流规则
    user = request.state.user
    if user.is_vip:
        return "100/minute"
    return "10/minute"

@app.get("/api/data")
@limiter.limit(dynamic_limit)
def get_data(request: Request):
    return {"data": []}
```

### 1.4 自定义限流响应

```python
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
from slowapi import Limiter
from slowapi.util import get_remote_address
from slowapi.errors import RateLimitExceeded

app = FastAPI()

limiter = Limiter(key_func=get_remote_address)
app.state.limiter = limiter


@app.exception_handler(RateLimitExceeded)
async def rate_limit_handler(request: Request, exc: RateLimitExceeded):
    """自定义限流响应"""
    return JSONResponse(
        status_code=429,
        content={
            "code": 429,
            "message": "请求过于频繁，请稍后再试",
            "detail": str(exc.detail)
        }
    )


@app.get("/limited")
@limiter.limit("5/minute")
def limited_api(request: Request):
    return {"message": "请求成功"}
```

### 1.5 基于用户的限流

```python
from fastapi import FastAPI, Request, Depends
from slowapi import Limiter
from slowapi.util import get_remote_address

app = FastAPI()


def get_user_id(request: Request) -> str:
    """
    获取用户标识（用于限流）
    
    优先级：
    1. 已登录用户的 ID
    2. API Key
    3. IP 地址
    """
    # 从请求中获取用户信息
    user = getattr(request.state, "user", None)
    if user:
        return f"user:{user.id}"
    
    # 从 Header 获取 API Key
    api_key = request.headers.get("X-API-Key")
    if api_key:
        return f"api_key:{api_key}"
    
    # 使用 IP 地址
    return f"ip:{get_remote_address(request)}"


limiter = Limiter(key_func=get_user_id)
app.state.limiter = limiter


@app.get("/api/data")
@limiter.limit("10/minute")
def get_data(request: Request):
    return {"data": []}
```

## 二、请求超时控制

### 2.1 使用 middleware 实现超时

```python
"""
请求超时控制
"""
import asyncio
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
from fastapi.exceptions import HTTPException

app = FastAPI()

# 超时时间（秒）
REQUEST_TIMEOUT = 30


@app.middleware("http")
async def timeout_middleware(request: Request, call_next):
    """请求超时中间件"""
    try:
        return await asyncio.wait_for(
            call_next(request),
            timeout=REQUEST_TIMEOUT
        )
    except asyncio.TimeoutError:
        return JSONResponse(
            status_code=504,
            content={
                "code": 504,
                "message": f"请求超时（超过 {REQUEST_TIMEOUT} 秒）"
            }
        )


@app.get("/slow")
async def slow_api():
    """模拟慢接口"""
    await asyncio.sleep(35)  # 超过超时时间
    return {"message": "响应成功"}
```

### 2.2 不同接口不同超时时间

```python
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
import asyncio

app = FastAPI()

# 默认超时时间
DEFAULT_TIMEOUT = 30

# 特定接口的超时时间
TIMEOUT_MAP = {
    "/api/export": 300,      # 导出接口 5 分钟
    "/api/upload": 120,      # 上传接口 2 分钟
    "/api/report": 60,       # 报表接口 1 分钟
}


@app.middleware("http")
async def dynamic_timeout_middleware(request: Request, call_next):
    """动态超时中间件"""
    # 获取超时时间
    timeout = TIMEOUT_MAP.get(request.url.path, DEFAULT_TIMEOUT)
    
    try:
        return await asyncio.wait_for(
            call_next(request),
            timeout=timeout
        )
    except asyncio.TimeoutError:
        return JSONResponse(
            status_code=504,
            content={
                "code": 504,
                "message": f"请求超时（超过 {timeout} 秒）"
            }
        )
```

## 三、输入验证与安全

### 3.1 SQL 注入防护

**使用 ORM（推荐）**：
```python
from sqlalchemy.orm import Session
from models import User

# 安全：ORM 自动转义
def get_user_by_username(db: Session, username: str):
    return db.query(User).filter(User.username == username).first()

# 不安全：直接拼接 SQL（不要这样做！）
# cursor.execute(f"SELECT * FROM users WHERE username = '{username}'")
```

### 3.2 XSS 防护

```python
from fastapi import FastAPI
from pydantic import BaseModel, field_validator
import html

app = FastAPI()


class CommentCreate(BaseModel):
    content: str
    
    @field_validator('content')
    @classmethod
    def sanitize_content(cls, v):
        """转义 HTML 特殊字符"""
        return html.escape(v)


@app.post("/comments")
def create_comment(comment: CommentCreate):
    # content 已被转义
    return {"content": comment.content}
```

### 3.3 文件上传安全

```python
from fastapi import FastAPI, UploadFile, File, HTTPException
from pathlib import Path
import magic  # pip install python-magic-bin (Windows) 或 python-magic (Linux)

app = FastAPI()

# 允许的 MIME 类型
ALLOWED_MIME_TYPES = {
    "image/jpeg",
    "image/png",
    "image/gif",
    "application/pdf"
}

# 最大文件大小（字节）
MAX_FILE_SIZE = 10 * 1024 * 1024  # 10MB


@app.post("/upload/secure")
async def secure_upload(file: UploadFile = File(...)):
    """安全的文件上传"""
    
    # 1. 检查文件大小
    contents = await file.read()
    if len(contents) > MAX_FILE_SIZE:
        raise HTTPException(status_code=413, detail="文件过大")
    
    # 2. 检查文件类型（通过文件内容，而非扩展名）
    mime_type = magic.from_buffer(contents, mime=True)
    if mime_type not in ALLOWED_MIME_TYPES:
        raise HTTPException(status_code=400, detail=f"不支持的文件类型: {mime_type}")
    
    # 3. 生成安全的文件名
    import uuid
    file_ext = Path(file.filename).suffix
    safe_filename = f"{uuid.uuid4()}{file_ext}"
    
    # 4. 保存文件
    file_path = Path("uploads") / safe_filename
    with open(file_path, "wb") as buffer:
        buffer.write(contents)
    
    return {
        "filename": safe_filename,
        "mime_type": mime_type,
        "size": len(contents)
    }
```

## 四、API Key 认证

### 4.1 简单 API Key

```python
from fastapi import FastAPI, Security, HTTPException
from fastapi.security import APIKeyHeader
from typing import Optional

app = FastAPI()

# API Key 配置
API_KEYS = {
    "key1": {"name": "Client 1", "rate_limit": 100},
    "key2": {"name": "Client 2", "rate_limit": 50}
}

api_key_header = APIKeyHeader(name="X-API-Key", auto_error=False)


async def get_api_key(api_key: Optional[str] = Security(api_key_header)):
    """验证 API Key"""
    if api_key is None:
        raise HTTPException(status_code=401, detail="缺少 API Key")
    
    if api_key not in API_KEYS:
        raise HTTPException(status_code=401, detail="无效的 API Key")
    
    return api_key


@app.get("/protected")
def protected_api(api_key: str = Depends(get_api_key)):
    """需要 API Key 的接口"""
    client = API_KEYS[api_key]
    return {"message": f"Hello, {client['name']}!"}
```

### 4.2 带权限的 API Key

```python
from fastapi import FastAPI, Security, HTTPException, Depends
from fastapi.security import APIKeyHeader
from typing import Optional, List

app = FastAPI()

# API Key 配置（带权限）
API_KEYS = {
    "admin_key": {"name": "Admin", "permissions": ["read", "write", "delete"]},
    "user_key": {"name": "User", "permissions": ["read"]},
    "readonly_key": {"name": "ReadOnly", "permissions": ["read"]}
}

api_key_header = APIKeyHeader(name="X-API-Key")


def require_permission(permission: str):
    """权限检查依赖"""
    async def check_permission(api_key: str = Security(api_key_header)):
        if api_key not in API_KEYS:
            raise HTTPException(status_code=401, detail="无效的 API Key")
        
        client = API_KEYS[api_key]
        if permission not in client["permissions"]:
            raise HTTPException(status_code=403, detail="无权限")
        
        return client
    
    return check_permission


@app.get("/data")
def read_data(client: dict = Depends(require_permission("read"))):
    """读取数据（需要 read 权限）"""
    return {"data": [], "client": client["name"]}


@app.post("/data")
def write_data(client: dict = Depends(require_permission("write"))):
    """写入数据（需要 write 权限）"""
    return {"message": "写入成功", "client": client["name"]}


@app.delete("/data/{id}")
def delete_data(id: int, client: dict = Depends(require_permission("delete"))):
    """删除数据（需要 delete 权限）"""
    return {"message": f"删除成功: {id}", "client": client["name"]}
```

## 五、IP 白名单

```python
from fastapi import FastAPI, Request, HTTPException
from fastapi.responses import JSONResponse

app = FastAPI()

# IP 白名单
IP_WHITELIST = [
    "127.0.0.1",
    "::1",
    "192.168.1.0/24",  # 支持 CIDR
]


def is_ip_allowed(ip: str) -> bool:
    """检查 IP 是否在白名单中"""
    import ipaddress
    
    for allowed in IP_WHITELIST:
        try:
            if "/" in allowed:
                # CIDR 格式
                network = ipaddress.ip_network(allowed, strict=False)
                if ipaddress.ip_address(ip) in network:
                    return True
            else:
                # 单个 IP
                if ip == allowed:
                    return True
        except ValueError:
            continue
    
    return False


@app.middleware("http")
async def ip_whitelist_middleware(request: Request, call_next):
    """IP 白名单中间件"""
    client_ip = request.client.host if request.client else None
    
    if client_ip and not is_ip_allowed(client_ip):
        return JSONResponse(
            status_code=403,
            content={
                "code": 403,
                "message": "IP 不在白名单中"
            }
        )
    
    return await call_next(request)


@app.get("/admin")
def admin_api():
    """管理员接口（仅白名单 IP 可访问）"""
    return {"message": "管理员接口"}
```

## 六、完整安全配置示例

```python
"""
完整的安全配置示例
"""
from fastapi import FastAPI, Request, Depends, Security, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from fastapi.security import APIKeyHeader
from fastapi.responses import JSONResponse
from slowapi import Limiter
from slowapi.util import get_remote_address
from slowapi.errors import RateLimitExceeded
import asyncio
from typing import Optional
import html

app = FastAPI(title="安全配置示例")

# ==================== CORS 配置 ====================

app.add_middleware(
    CORSMiddleware,
    allow_origins=["https://example.com"],  # 生产环境限制域名
    allow_credentials=True,
    allow_methods=["GET", "POST", "PUT", "DELETE"],
    allow_headers=["*"],
)

# ==================== 限流配置 ====================

limiter = Limiter(key_func=get_remote_address)
app.state.limiter = limiter


@app.exception_handler(RateLimitExceeded)
async def rate_limit_handler(request: Request, exc: RateLimitExceeded):
    return JSONResponse(
        status_code=429,
        content={"code": 429, "message": "请求过于频繁"}
    )

# ==================== 超时配置 ====================

REQUEST_TIMEOUT = 30


@app.middleware("http")
async def timeout_middleware(request: Request, call_next):
    try:
        return await asyncio.wait_for(call_next(request), timeout=REQUEST_TIMEOUT)
    except asyncio.TimeoutError:
        return JSONResponse(
            status_code=504,
            content={"code": 504, "message": "请求超时"}
        )

# ==================== 安全头 ====================

@app.middleware("http")
async def add_security_headers(request: Request, call_next):
    response = await call_next(request)
    
    # 防止 XSS
    response.headers["X-Content-Type-Options"] = "nosniff"
    response.headers["X-Frame-Options"] = "DENY"
    response.headers["X-XSS-Protection"] = "1; mode=block"
    
    # HTTPS 强制
    response.headers["Strict-Transport-Security"] = "max-age=31536000; includeSubDomains"
    
    return response

# ==================== API Key 认证 ====================

API_KEYS = {
    "test_key": {"name": "Test Client", "permissions": ["read", "write"]}
}

api_key_header = APIKeyHeader(name="X-API-Key")


async def get_api_key(api_key: Optional[str] = Security(api_key_header)):
    if api_key is None or api_key not in API_KEYS:
        raise HTTPException(status_code=401, detail="无效的 API Key")
    return API_KEYS[api_key]

# ==================== 接口 ====================

@app.get("/")
@limiter.limit("10/minute")
def root(request: Request):
    return {"message": "Hello World"}


@app.get("/protected")
@limiter.limit("5/minute")
def protected_api(
    request: Request,
    client: dict = Depends(get_api_key)
):
    return {"message": f"Hello, {client['name']}!"}
```
