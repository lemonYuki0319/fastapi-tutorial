## 一、配置管理简介

### 1.1 为什么需要配置管理？

**问题场景**：
- 开发环境、测试环境、生产环境配置不同
- 数据库密码、API 密钥等敏感信息不能硬编码
- 不同环境需要不同的日志级别、调试模式

**解决方案**：
- 使用 `.env` 文件管理配置
- 使用 `pydantic-settings` 进行配置校验
- 环境变量覆盖配置文件

### 2.4 加载配置

```python
"""
配置管理 - 使用 python-dotenv
"""
import os
from dotenv import load_dotenv

# 加载 .env 文件
load_dotenv()


class Settings:
    """配置类"""
    
    # 应用配置
    APP_NAME: str = os.getenv("APP_NAME", "FastAPI App")
    APP_VERSION: str = os.getenv("APP_VERSION", "1.0.0")
    DEBUG: bool = os.getenv("DEBUG", "True").lower() == "true"
    ENVIRONMENT: str = os.getenv("ENVIRONMENT", "development")
    
    # 服务配置
    HOST: str = os.getenv("HOST", "127.0.0.1")
    PORT: int = int(os.getenv("PORT", "8000"))
    
    # 数据库配置
    DATABASE_URL: str = os.getenv("DATABASE_URL", "sqlite:///./test_api.db")
    
    # JWT 配置
    SECRET_KEY: str = os.getenv("SECRET_KEY", "your-secret-key")
    ALGORITHM: str = os.getenv("ALGORITHM", "HS256")
    ACCESS_TOKEN_EXPIRE_MINUTES: int = int(os.getenv("ACCESS_TOKEN_EXPIRE_MINUTES", "30"))
    
    # 日志配置
    LOG_LEVEL: str = os.getenv("LOG_LEVEL", "INFO")
    LOG_FILE: str = os.getenv("LOG_FILE", "logs/app.log")


# 创建配置实例
settings = Settings()
```

## 三、使用 pydantic-settings（推荐）

### 3.1 安装依赖

```bash
pip install pydantic-settings
```

### 3.2 创建配置文件

```python
"""
配置管理 - 使用 pydantic-settings
推荐方式：类型安全、自动校验
"""
from pydantic_settings import BaseSettings, SettingsConfigDict
from typing import List
from functools import lru_cache


class Settings(BaseSettings):
    """
    应用配置类
    
    使用 pydantic-settings 自动从环境变量和 .env 文件加载配置
    """
    
    # 应用配置
    APP_NAME: str = "FastAPI CRUD Tutorial"
    APP_VERSION: str = "1.0.0"
    DEBUG: bool = False
    ENVIRONMENT: str = "development"
    
    # 服务配置
    HOST: str = "127.0.0.1"
    PORT: int = 8000
    
    # 数据库配置
    DATABASE_URL: str = "sqlite:///./test_api.db"
    DATABASE_POOL_SIZE: int = 5
    DATABASE_MAX_OVERFLOW: int = 10
    
    # JWT 配置
    SECRET_KEY: str = "your-secret-key-change-in-production"
    ALGORITHM: str = "HS256"
    ACCESS_TOKEN_EXPIRE_MINUTES: int = 30
    
    # 日志配置
    LOG_LEVEL: str = "INFO"
    LOG_FILE: str = "logs/app.log"
    LOG_ROTATION: str = "10 MB"
    LOG_RETENTION: str = "7 days"
    
    # CORS 配置
    CORS_ORIGINS: List[str] = ["*"]
    CORS_ALLOW_CREDENTIALS: bool = True
    CORS_ALLOW_METHODS: List[str] = ["*"]
    CORS_ALLOW_HEADERS: List[str] = ["*"]
    
    # 限流配置
    RATE_LIMIT_REQUESTS: int = 100
    RATE_LIMIT_PERIOD: int = 60  # 秒
    
    # 配置模型
    model_config = SettingsConfigDict(
        env_file=".env",           # .env 文件路径
        env_file_encoding="utf-8", # 文件编码
        case_sensitive=True,       # 区分大小写
        extra="ignore"             # 忽略额外字段
    )


# 使用缓存装饰器，确保只创建一个配置实例
@lru_cache()
def get_settings() -> Settings:
    """
    获取配置实例（单例模式）
    
    使用 lru_cache 确保配置只加载一次
    """
    return Settings()


# 使用示例
settings = get_settings()
```

### 3.3 在代码中使用配置

```python
from fastapi import FastAPI, Depends
from config import get_settings, Settings

app = FastAPI()


# 方式一：直接使用
@app.get("/info")
def get_info():
    return {
        "app_name": settings.APP_NAME,
        "version": settings.APP_VERSION,
        "environment": settings.ENVIRONMENT
    }


# 方式二：依赖注入（推荐，方便测试）
@app.get("/config")
def get_config(settings: Settings = Depends(get_settings)):
    return {
        "app_name": settings.APP_NAME,
        "debug": settings.DEBUG
    }
```

### 4.4 根据环境加载配置

```python
"""
配置管理 - 多环境支持
"""
import os
from pydantic_settings import BaseSettings, SettingsConfigDict
from functools import lru_cache


class Settings(BaseSettings):
    """应用配置类"""
    
    APP_NAME: str = "FastAPI App"
    DEBUG: bool = False
    ENVIRONMENT: str = "development"
    DATABASE_URL: str = "sqlite:///./test_api.db"
    SECRET_KEY: str = "your-secret-key"
    
    model_config = SettingsConfigDict(
        env_file=".env",  # 默认文件
        env_file_encoding="utf-8",
        case_sensitive=True,
        extra="ignore"
    )


def get_env_file() -> str:
    """
    根据环境变量选择配置文件
    
    环境变量 ENVIRONMENT 的值：
    - development: .env.development
    - testing: .env.testing
    - production: .env.production
    """
    env = os.getenv("ENVIRONMENT", "development")
    env_files = {
        "development": ".env.development",
        "testing": ".env.testing",
        "production": ".env.production"
    }
    return env_files.get(env, ".env")


class DynamicSettings(BaseSettings):
    """动态加载配置的配置类"""
    
    APP_NAME: str = "FastAPI App"
    DEBUG: bool = False
    ENVIRONMENT: str = "development"
    DATABASE_URL: str = "sqlite:///./test_api.db"
    SECRET_KEY: str = "your-secret-key"
    
    model_config = SettingsConfigDict(
        env_file=get_env_file(),
        env_file_encoding="utf-8",
        case_sensitive=True,
        extra="ignore"
    )


@lru_cache()
def get_settings() -> DynamicSettings:
    """获取配置实例"""
    return DynamicSettings()


settings = get_settings()
```

### 6.2 完整配置文件

```python
"""
配置管理 - 完整示例
"""
import os
from typing import List
from functools import lru_cache
from pydantic_settings import BaseSettings, SettingsConfigDict
from pydantic import Field, validator


class DatabaseSettings(BaseSettings):
    """数据库配置"""
    
    URL: str = Field(default="sqlite:///./test_api.db", alias="DATABASE_URL")
    POOL_SIZE: int = Field(default=5, alias="DATABASE_POOL_SIZE")
    MAX_OVERFLOW: int = Field(default=10, alias="DATABASE_MAX_OVERFLOW")
    ECHO: bool = Field(default=False, alias="DATABASE_ECHO")
    
    model_config = SettingsConfigDict(env_prefix="DATABASE_")


class JWTSettings(BaseSettings):
    """JWT 配置"""
    
    SECRET_KEY: str = Field(default="your-secret-key", alias="JWT_SECRET_KEY")
    ALGORITHM: str = Field(default="HS256", alias="JWT_ALGORITHM")
    ACCESS_TOKEN_EXPIRE_MINUTES: int = Field(default=30, alias="JWT_ACCESS_TOKEN_EXPIRE_MINUTES")
    
    model_config = SettingsConfigDict(env_prefix="JWT_")


class LogSettings(BaseSettings):
    """日志配置"""
    
    LEVEL: str = Field(default="INFO", alias="LOG_LEVEL")
    FILE: str = Field(default="logs/app.log", alias="LOG_FILE")
    ROTATION: str = Field(default="10 MB", alias="LOG_ROTATION")
    RETENTION: str = Field(default="7 days", alias="LOG_RETENTION")
    
    model_config = SettingsConfigDict(env_prefix="LOG_")


class Settings(BaseSettings):
    """
    应用配置类
    
    配置加载优先级：
    1. 环境变量（最高优先级）
    2. .env 文件
    3. 默认值
    """
    
    # 应用配置
    APP_NAME: str = "FastAPI CRUD Tutorial"
    APP_VERSION: str = "1.0.0"
    APP_DESCRIPTION: str = "FastAPI CRUD 实战教程"
    DEBUG: bool = False
    ENVIRONMENT: str = "development"
    
    # 服务配置
    HOST: str = "127.0.0.1"
    PORT: int = 8000
    WORKERS: int = 1
    
    # 子配置
    DATABASE: DatabaseSettings = DatabaseSettings()
    JWT: JWTSettings = JWTSettings()
    LOG: LogSettings = LogSettings()
    
    # CORS 配置
    CORS_ORIGINS: List[str] = ["*"]
    CORS_ALLOW_CREDENTIALS: bool = True
    CORS_ALLOW_METHODS: List[str] = ["*"]
    CORS_ALLOW_HEADERS: List[str] = ["*"]
    
    # 限流配置
    RATE_LIMIT_ENABLED: bool = True
    RATE_LIMIT_REQUESTS: int = 100
    RATE_LIMIT_PERIOD: int = 60
    
    # 验证器
    @validator("ENVIRONMENT")
    def validate_environment(cls, v):
        allowed = ["development", "testing", "production"]
        if v not in allowed:
            raise ValueError(f"ENVIRONMENT 必须是 {allowed} 之一")
        return v
    
    @validator("CORS_ORIGINS", pre=True)
    def parse_cors_origins(cls, v):
        if isinstance(v, str):
            # 解析 JSON 字符串
            import json
            return json.loads(v)
        return v
    
    # 配置模型
    model_config = SettingsConfigDict(
        env_file=".env",
        env_file_encoding="utf-8",
        case_sensitive=True,
        extra="ignore"
    )


@lru_cache()
def get_settings() -> Settings:
    """
    获取配置实例（单例）
    
    使用 lru_cache 确保配置只加载一次
    """
    return Settings()


# 创建全局配置实例
settings = get_settings()
```

### 6.3 在应用中使用

```python
"""
主程序 - 使用配置
"""
from fastapi import FastAPI, Depends
from fastapi.middleware.cors import CORSMiddleware
from config import get_settings, Settings
from database import engine, Base

# 创建数据库表
Base.metadata.create_all(bind=engine)

# 创建应用
settings = get_settings()

app = FastAPI(
    title=settings.APP_NAME,
    description=settings.APP_DESCRIPTION,
    version=settings.APP_VERSION,
    debug=settings.DEBUG
)

# CORS 配置
app.add_middleware(
    CORSMiddleware,
    allow_origins=settings.CORS_ORIGINS,
    allow_credentials=settings.CORS_ALLOW_CREDENTIALS,
    allow_methods=settings.CORS_ALLOW_METHODS,
    allow_headers=settings.CORS_ALLOW_HEADERS,
)


@app.get("/")
def root():
    """根路径"""
    return {
        "app": settings.APP_NAME,
        "version": settings.APP_VERSION,
        "environment": settings.ENVIRONMENT
    }


@app.get("/config")
def get_config(settings: Settings = Depends(get_settings)):
    """获取配置信息（仅开发环境）"""
    if settings.ENVIRONMENT == "production":
        return {"error": "生产环境不允许查看配置"}
    
    return {
        "app_name": settings.APP_NAME,
        "debug": settings.DEBUG,
        "environment": settings.ENVIRONMENT,
        "database_url": settings.DATABASE.URL,
        "log_level": settings.LOG.LEVEL
    }
```

## 八、常见问题

### Q1: .env 文件不生效？

检查以下几点：
1. `.env` 文件是否在项目根目录
2. 文件名是否正确（注意没有前缀）
3. 格式是否正确（`KEY=VALUE`，没有空格）
4. 是否在代码中调用了 `load_dotenv()`

### Q2: 如何在 Docker 中使用？

```dockerfile
# Dockerfile
FROM python:3.9

WORKDIR /app

COPY . .

RUN pip install -r requirements.txt

# 不复制 .env 文件，使用环境变量
CMD ["uvicorn", "main:app", "--host", "0.0.0.0", "--port", "8000"]
```

```yaml
# docker-compose.yml
version: '3.8'

services:
  app:
    build: .
    ports:
      - "8000:8000"
    environment:
      - APP_NAME=FastAPI App
      - DATABASE_URL=sqlite:///./test_api.db
      - SECRET_KEY=your-secret-key
```

### Q3: 如何验证配置是否正确？

```python
from config import get_settings

settings = get_settings()

# 验证必需的配置
assert settings.JWT.SECRET_KEY != "your-secret-key", "请修改 SECRET_KEY"
assert settings.DATABASE.URL, "DATABASE_URL 不能为空"
```
