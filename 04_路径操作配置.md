## 一、路径操作装饰器参数

### 1.1 完整参数列表

```python
from fastapi import FastAPI, status
from pydantic import BaseModel

app = FastAPI()

class Item(BaseModel):
    name: str
    price: float


@app.post(
    "/items",
    
    # 响应配置
    response_model=Item,                    # 响应模型
    response_model_exclude_unset=True,      # 排除未设置的字段
    response_model_exclude_none=True,       # 排除 None 值
    response_model_include={"name", "price"},  # 只包含指定字段
    response_model_exclude={"secret_field"},   # 排除指定字段
    
    # 状态码
    status_code=status.HTTP_201_CREATED,    # 成功状态码
    
    # 文档配置
    tags=["商品管理"],                       # 标签（分组）
    summary="创建商品",                      # 摘要
    description="创建一个新商品，返回创建的商品信息",  # 描述
    response_description="创建成功的商品信息",  # 响应描述
    
    # 响应配置
    responses={
        201: {"description": "创建成功"},
        400: {"description": "参数错误"},
        409: {"description": "商品已存在"}
    },
    
    # 其他配置
    deprecated=False,                        # 是否废弃
    operation_id="create_item",              # 操作 ID（用于 OpenAPI）
    
    # 安全配置
    # security=[{"OAuth2PasswordBearer": []}]
)
def create_item(item: Item):
    """创建商品"""
    return item
```

## 二、状态码配置

### 2.1 使用 status 模块

```python
from fastapi import FastAPI, status, HTTPException

app = FastAPI()

# 创建资源：201 Created
@app.post("/users", status_code=status.HTTP_201_CREATED)
def create_user():
    return {"message": "用户创建成功"}


# 删除资源：204 No Content
@app.delete("/users/{user_id}", status_code=status.HTTP_204_NO_CONTENT)
def delete_user(user_id: int):
    return None  # 204 不返回内容


# 更新资源：200 OK
@app.put("/users/{user_id}", status_code=status.HTTP_200_OK)
def update_user(user_id: int):
    return {"message": "用户更新成功"}


# 自定义状态码
@app.get("/custom-status")
def custom_status():
    # 可以在函数内返回不同状态码
    from fastapi import Response
    return Response(content="自定义状态码", status_code=202)
```

### 2.2 常用状态码

| 状态码 | 说明 | 使用场景 |
|--------|------|---------|
| 200 | OK | 成功（默认） |
| 201 | Created | 创建成功 |
| 204 | No Content | 删除成功（无返回内容） |
| 400 | Bad Request | 客户端参数错误 |
| 401 | Unauthorized | 未认证 |
| 403 | Forbidden | 无权限 |
| 404 | Not Found | 资源不存在 |
| 409 | Conflict | 资源冲突 |
| 422 | Unprocessable Entity | 参数校验失败 |
| 500 | Internal Server Error | 服务器错误 |

## 三、标签与分组

### 3.1 使用标签分组

```python
from fastapi import FastAPI

app = FastAPI()

# 用户管理标签
@app.get("/users", tags=["用户管理"])
def get_users():
    """查询用户列表"""
    return [{"id": 1, "name": "用户1"}]


@app.post("/users", tags=["用户管理"])
def create_user():
    """创建用户"""
    return {"message": "创建成功"}


# 商品管理标签
@app.get("/products", tags=["商品管理"])
def get_products():
    """查询商品列表"""
    return [{"id": 1, "name": "商品1"}]


@app.post("/products", tags=["商品管理"])
def create_product():
    """创建商品"""
    return {"message": "创建成功"}


# 订单管理标签
@app.get("/orders", tags=["订单管理"])
def get_orders():
    """查询订单列表"""
    return [{"id": 1, "order_no": "ORDER001"}]
```

**效果**：在 Swagger UI 中，接口会按标签分组显示。

### 3.2 多标签

```python
# 一个接口可以有多个标签
@app.get("/reports", tags=["报表管理", "数据分析"])

    """查询报表（同时属于两个分组）"""
```

## 四、摘要与描述

### 4.1 摘要（summary）

```python
@app.get("/users/{user_id}", summary="查询单个用户")
def get_user(user_id: int):
    """根据用户ID查询用户详细信息"""
    return {"id": user_id, "name": "用户"}
```

### 4.2 描述（description）

**方式一：字符串描述**

```python
@app.get(
    "/items/{item_id}",
    summary="查询商品",
    description="""
## 查询商品接口

### 功能说明
根据商品ID查询商品详细信息。

### 参数说明
- **item_id**: 商品ID，必须大于0

### 返回值
返回商品的完整信息，包括名称、价格、库存等。

### 错误码
- 404: 商品不存在
- 422: 参数校验失败
"""
)
def get_item(item_id: int):
    return {"id": item_id}
```

**方式二：文档字符串**

```python
@app.get("/items/{item_id}")
def get_item(item_id: int):
    """
    查询商品接口
    
    根据商品ID查询商品详细信息。
    
    - **item_id**: 商品ID
    """
    return {"id": item_id}
```

### 4.3 响应描述

```python
@app.post(

创建成功的用户信息，包含：
- id: 用户ID
```

## 五、响应模型配置

### 5.1 response_model

```python
from pydantic import BaseModel
from typing import Optional
from datetime import datetime

class UserResponse(BaseModel):
    id: int
    username: str
    email: str
    password: str  # 敏感字段
    created_at: datetime


@app.get("/users/{user_id}", response_model=UserResponse)
def get_user(user_id: int):
    # 返回的数据会自动按 UserResponse 格式化
    return {
        "id": user_id,
        "username": "test",
        "email": "test@example.com",
        "password": "secret",  # 不应该返回
        "created_at": datetime.now()
    }
```

### 5.2 排除敏感字段

```python
# 方式一：response_model_exclude
@app.get("/users/{user_id}", response_model=UserResponse, response_model_exclude={"password"})
def get_user(user_id: int):
    return {"id": user_id, "username": "test", "email": "test@example.com", "password": "secret", "created_at": datetime.now()}


# 方式二：response_model_include
@app.get("/users/{user_id}", response_model=UserResponse, response_model_include={"id", "username", "email"})
def get_user(user_id: int):
    return {"id": user_id, "username": "test", "email": "test@example.com", "password": "secret", "created_at": datetime.now()}


# 方式三：排除未设置的字段
class UserResponse(BaseModel):
    id: int
    username: str
    email: Optional[str] = None
    phone: Optional[str] = None


@app.get("/users/{user_id}", response_model=UserResponse, response_model_exclude_unset=True)
def get_user(user_id: int):
    # 只返回有值的字段，None 字段会被排除
    return {"id": user_id, "username": "test", "email": None, "phone": None}
```

## 六、响应头与 Cookie

### 6.1 设置响应头

```python
from fastapi import FastAPI, Response

app = FastAPI()

@app.get("/headers")
def get_with_headers(response: Response):
    """设置响应头"""
    response.headers["X-Custom-Header"] = "Custom Value"
    response.headers["X-Request-ID"] = "12345"
    return {"message": "查看响应头"}


# 使用 Response 对象
from fastapi.responses import JSONResponse

@app.get("/custom-response")
def custom_response():
    """自定义响应"""
    return JSONResponse(
        content={"message": "自定义响应"},
        headers={
            "X-Custom-Header": "Custom Value",
            "X-Another-Header": "Another Value"
        }
    )
```

### 6.2 设置 Cookie

```python
from fastapi import FastAPI, Response

app = FastAPI()

@app.post("/login")
def login(response: Response):
    """登录并设置 Cookie"""
    response.set_cookie(
        key="session_id",
        value="abc123",
        httponly=True,      # 防止 JavaScript 访问
        max_age=3600,       # 过期时间（秒）
        expires=3600,       # 过期时间
        path="/",           # Cookie 路径
        domain=None,        # Cookie 域名
        secure=False,       # 仅 HTTPS
        samesite="lax"      # 同站策略
    )
    return {"message": "登录成功"}


@app.get("/logout")
def logout(response: Response):
    """登出并删除 Cookie"""
    response.delete_cookie("session_id")
    return {"message": "登出成功"}
```

### 6.3 读取 Cookie

```python
from fastapi import FastAPI, Cookie

app = FastAPI()

@app.get("/profile")
def get_profile(session_id: str = Cookie(None)):
    """读取 Cookie"""
    if not session_id:
        return {"message": "未登录"}
    return {"session_id": session_id, "message": "已登录"}
```

## 七、多响应类型

### 7.1 定义多种响应

```python
from fastapi import FastAPI
from pydantic import BaseModel
from typing import Union

app = FastAPI()

class SuccessResponse(BaseModel):
    code: int = 200
    message: str
    data: dict


class ErrorResponse(BaseModel):
    code: int
    message: str
    detail: str


@app.get(
    "/items/{item_id}",
    response_model=Union[SuccessResponse, ErrorResponse],
    responses={
        200: {
            "description": "查询成功",
            "model": SuccessResponse
        },
        404: {
            "description": "商品不存在",
            "model": ErrorResponse
        }
    }
)
def get_item(item_id: int):
    if item_id == 0:
        return ErrorResponse(code=404, message="商品不存在", detail="ID不能为0")
    return SuccessResponse(message="查询成功", data={"id": item_id})
```

### 7.2 多状态码响应

## 九、完整示例

```python
from fastapi import FastAPI, status, Response
from pydantic import BaseModel, Field
from typing import Optional
from datetime import datetime

app = FastAPI(
    title="商品管理系统",
    description="商品 CRUD 接口示例",
    version="1.0.0"
)


# 模型定义
class ProductCreate(BaseModel):
    """创建商品请求模型"""
    name: str = Field(..., min_length=1, max_length=100, description="商品名称")
    price: float = Field(..., gt=0, description="商品价格")
    stock: int = Field(default=0, ge=0, description="库存数量")
    description: Optional[str] = Field(None, description="商品描述")


class ProductResponse(BaseModel):
    """商品响应模型"""
    id: int
    name: str
    price: float
    stock: int
    description: Optional[str]
    created_at: datetime
    
    class Config:
        json_schema_extra = {
            "example": {
                "id": 1,
                "name": "测试商品",
                "price": 99.9,
                "stock": 100,
                "description": "这是一个测试商品",
                "created_at": "2024-01-15T10:30:00"
            }
        }


# 模拟数据库
fake_db = {}
product_id_counter = 1


# 创建商品
@app.post(
    "/products",
    response_model=ProductResponse,
    status_code=status.HTTP_201_CREATED,
    tags=["商品管理"],
    summary="创建商品",
    description="""
## 创建商品接口

### 功能
创建一个新的商品。

### 参数
- **name**: 商品名称（必填，1-100字符）
- **price**: 商品价格（必填，大于0）
- **stock**: 库存数量（可选，默认0）
- **description**: 商品描述（可选）

### 返回
返回创建的商品信息，包含自动生成的ID和创建时间。
""",
    response_description="创建成功的商品信息",
    responses={
        201: {"description": "创建成功"},
        400: {"description": "参数错误"}
    }
)
def create_product(product: ProductCreate, response: Response):
    """创建商品"""
    global product_id_counter
    
    product_data = {
        "id": product_id_counter,
        "name": product.name,
        "price": product.price,
        "stock": product.stock,
        "description": product.description,
        "created_at": datetime.now()
    }
    
    fake_db[product_id_counter] = product_data
    
    # 设置响应头
    response.headers["X-Product-ID"] = str(product_id_counter)
    response.headers["Location"] = f"/products/{product_id_counter}"
    
    product_id_counter += 1
    return product_data


# 查询商品
@app.get(
    "/products/{product_id}",
    response_model=ProductResponse,
    tags=["商品管理"],
    summary="查询商品",
    description="根据商品ID查询商品详细信息",
    responses={
        200: {"description": "查询成功"},
        404: {"description": "商品不存在"}
    }
)
def get_product(product_id: int):
    """查询商品"""
    if product_id not in fake_db:
        from fastapi import HTTPException
        raise HTTPException(status_code=404, detail="商品不存在")
    return fake_db[product_id]


# 查询商品列表
@app.get(
    "/products",
    response_model=list[ProductResponse],
    tags=["商品管理"],
    summary="查询商品列表",
    description="查询所有商品列表"
)
def get_products(skip: int = 0, limit: int = 10):
    """查询商品列表"""
    products = list(fake_db.values())[skip:skip+limit]
    return products
```
