## 一、依赖注入简介

### 1.1 什么是依赖注入？

依赖注入（Dependency Injection, DI）是一种设计模式，用于将组件的依赖关系从组件内部移到外部管理。

**FastAPI 中的依赖注入**：
- 自动执行某些代码，在接口处理前
- 获取某些值或对象
- 实现代码复用
- 类似于装饰器，但更强大

### 1.2 为什么使用依赖注入？

**场景一：数据库连接**

```python
# 不使用依赖注入（每个接口都要写）
@app.get("/users")
def get_users():
    db = SessionLocal()  # 创建连接
    try:
        users = db.query(User).all()
        return users
    finally:
        db.close()  # 关闭连接（容易忘记）


# 使用依赖注入（自动管理）
@app.get("/users")
def get_users(db: Session = Depends(get_db)):
    users = db.query(User).all()
    return users
    # 自动关闭连接
```

**场景二：公共参数校验**

```python
# 不使用依赖注入（每个接口都要写）
@app.get("/items")
def get_items(skip: int = 0, limit: int = 10):
    if limit > 100:
        limit = 100
    # ...

@app.get("/users")
def get_users(skip: int = 0, limit: int = 10):
    if limit > 100:
        limit = 100
    # ...


# 使用依赖注入（统一处理）
class Pagination:
    def __call__(self, skip: int = 0, limit: int = Query(10, le=100)):
        return {"skip": skip, "limit": limit}

pagination = Pagination()

@app.get("/items")
def get_items(pag: dict = Depends(pagination)):
    # pag = {"skip": 0, "limit": 10}
    pass
```

## 二、基本用法

### 2.1 函数作为依赖

```python
from fastapi import FastAPI, Depends

app = FastAPI()


# 定义依赖函数
def common_parameters(skip: int = 0, limit: int = 10):
    """公共分页参数"""
    return {"skip": skip, "limit": limit}


# 使用依赖
@app.get("/users")
def get_users(params: dict = Depends(common_parameters)):
    """查询用户列表"""
    return {
        "skip": params["skip"],
        "limit": params["limit"],
        "users": ["user1", "user2"]
    }


@app.get("/items")
def get_items(params: dict = Depends(common_parameters)):
    """查询商品列表"""
    return {
        "skip": params["skip"],
        "limit": params["limit"],
        "items": ["item1", "item2"]
    }
```

### 2.2 类作为依赖

```python
from fastapi import FastAPI, Depends, Query

app = FastAPI()


# 定义依赖类
class Pagination:
    """分页参数类"""
    def __init__(
        self,
        skip: int = Query(0, ge=0, description="跳过记录数"),
        limit: int = Query(10, ge=1, le=100, description="返回记录数")
    ):
        self.skip = skip
        self.limit = limit
    
    def get_offset(self):
        """计算偏移量"""
        return self.skip
    
    def get_limit(self):
        """获取限制数"""
        return self.limit


# 使用依赖
@app.get("/users")
def get_users(pag: Pagination = Depends()):
    """查询用户列表"""
    return {
        "skip": pag.skip,
        "limit": pag.limit,
        "offset": pag.get_offset()
    }
```

### 2.3 依赖的返回值

```python
from fastapi import FastAPI, Depends

app = FastAPI()


def get_db():
    """数据库连接依赖"""
    db = "Database Connection"
    try:
        yield db  # 使用 yield，会在请求结束后执行 finally
    finally:
        print("关闭数据库连接")


@app.get("/users")
def get_users(db = Depends(get_db)):
    """查询用户"""
    return {"db": db}
```

## 三、数据库连接依赖

### 3.1 完整示例

```python
from fastapi import FastAPI, Depends
from sqlalchemy import create_engine, Column, Integer, String
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker, Session

# 数据库配置
SQLALCHEMY_DATABASE_URL = "sqlite:///./test.db"
engine = create_engine(SQLALCHEMY_DATABASE_URL, connect_args={"check_same_thread": False})
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)
Base = declarative_base()


# 定义模型
class User(Base):
    __tablename__ = "users"
    id = Column(Integer, primary_key=True, index=True)
    username = Column(String(50), unique=True)
    email = Column(String(100))


# 创建表
Base.metadata.create_all(bind=engine)

app = FastAPI()


# 数据库依赖
def get_db():
    """获取数据库会话"""
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()


# 使用依赖
@app.get("/users")
def get_users(db: Session = Depends(get_db)):
    """查询用户列表"""
    users = db.query(User).all()
    return users


@app.post("/users")
def create_user(username: str, email: str, db: Session = Depends(get_db)):
    """创建用户"""
    user = User(username=username, email=email)
    db.add(user)
    db.commit()
    db.refresh(user)
    return user
```

## 四、依赖层级

### 4.1 依赖嵌套

依赖可以依赖其他依赖，形成依赖树。

```python
from fastapi import FastAPI, Depends, Header, HTTPException

app = FastAPI()


# 第一层依赖：解析 Token
def parse_token(authorization: str = Header(...)):
    """解析 Authorization 头"""
    if not authorization.startswith("Bearer "):
        raise HTTPException(status_code=401, detail="无效的认证格式")
    token = authorization.replace("Bearer ", "")
    return token


# 第二层依赖：验证 Token（依赖第一层）
def verify_token(token: str = Depends(parse_token)):
    """验证 Token"""
    # 模拟验证
    if token != "valid_token":
        raise HTTPException(status_code=401, detail="Token 无效")
    return {"user_id": 1, "username": "admin"}


# 第三层依赖：获取当前用户（依赖第二层）
def get_current_user(user_info: dict = Depends(verify_token)):
    """获取当前用户"""
    return {"id": user_info["user_id"], "username": user_info["username"]}


# 使用依赖
@app.get("/profile")
def get_profile(current_user: dict = Depends(get_current_user)):
    """获取用户信息（需要认证）"""
    return {
        "message": "获取成功",
        "user": current_user
    }
```

**依赖树**：
```
get_profile
    └── get_current_user
            └── verify_token
                    └── parse_token
```

## 五、全局依赖

### 5.1 应用级全局依赖

```python
from fastapi import FastAPI, Depends, Request
from time import time

app = FastAPI()


# 定义依赖
def log_request(request: Request):
    """记录请求日志"""
    print(f"[{time()}] {request.method} {request.url}")
    return request


# 应用级全局依赖
app = FastAPI(dependencies=[Depends(log_request)])


@app.get("/users")
def get_users():
    """所有接口都会自动执行 log_request"""
    return {"users": []}
```

### 5.2 路由级全局依赖

```python
from fastapi import FastAPI, Depends, HTTPException, Header
from typing import Optional

app = FastAPI()


# 定义认证依赖
def verify_api_key(api_key: Optional[str] = Header(None)):
    """验证 API Key"""
    if api_key != "secret_key":
        raise HTTPException(status_code=401, detail="无效的 API Key")
    return api_key


# 创建路由器（带全局依赖）
from fastapi import APIRouter

protected_router = APIRouter(
    prefix="/protected",
    dependencies=[Depends(verify_api_key)],
    tags=["需要认证的接口"]
)


# 路由级依赖：所有 /protected 下的接口都需要认证
@protected_router.get("/users")
def get_users():
    """查询用户（需要认证）"""
    return {"users": []}


@protected_router.get("/items")
def get_items():
    """查询商品（需要认证）"""
    return {"items": []}


# 注册路由
app.include_router(protected_router)


# 公开接口（不需要认证）
@app.get("/public")
def public_api():
    """公开接口"""
    return {"message": "公开接口"}
```

## 六、可调用对象作为依赖

### 6.1 使用 __call__

```python
from fastapi import FastAPI, Depends, Query

app = FastAPI()


class Pagination:
    """分页依赖类"""
    def __init__(self, max_limit: int = 100):
        self.max_limit = max_limit
    
    def __call__(
        self,
        skip: int = Query(0, ge=0),
        limit: int = Query(10, ge=1)
    ):
        # 限制最大值
        if limit > self.max_limit:
            limit = self.max_limit
        return {"skip": skip, "limit": limit}


# 创建实例
pagination = Pagination(max_limit=50)


@app.get("/users")
def get_users(pag: dict = Depends(pagination)):
    """查询用户"""
    return pag


@app.get("/items")
def get_items(pag: dict = Depends(pagination)):
    """查询商品"""
    return pag
```

### 6.2 使用 Depends() 自动实例化

```python
class Pagination:
    """分页依赖类"""
    def __init__(
        self,
        skip: int = Query(0, ge=0),
        limit: int = Query(10, ge=1, le=100)
    ):
        self.skip = skip
        self.limit = limit


@app.get("/users")
def get_users(pag: Pagination = Depends()):
    # 等同于 pag: Pagination = Depends(Pagination)
    return {"skip": pag.skip, "limit": pag.limit}
```

## 七、依赖缓存

### 7.1 默认缓存行为

默认情况下，同一请求中多次使用同一依赖，只会执行一次。

```python
from fastapi import FastAPI, Depends

app = FastAPI()

call_count = 0


def dependency():
    """计数依赖"""
    global call_count
    call_count += 1
    print(f"依赖被调用 {call_count} 次")
    return call_count


@app.get("/test")
def test(dep1: int = Depends(dependency), dep2: int = Depends(dependency)):
    """
    同一请求中，dependency 只执行一次
    dep1 和 dep2 的值相同
    """
    return {"dep1": dep1, "dep2": dep2}
```

### 7.2 禁用缓存

```python
@app.get("/test-no-cache")
def test_no_cache(
    dep1: int = Depends(dependency, use_cache=False),
    dep2: int = Depends(dependency, use_cache=False)
):
    """
    禁用缓存后，dependency 执行两次
    dep1 和 dep2 的值不同
    """
    return {"dep1": dep1, "dep2": dep2}
```

## 八、完整示例：用户认证系统

```python
from fastapi import FastAPI, Depends, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from pydantic import BaseModel
from typing import Optional
from datetime import datetime, timedelta
import jwt

app = FastAPI()

# JWT 配置
SECRET_KEY = "your-secret-key"
ALGORITHM = "HS256"
ACCESS_TOKEN_EXPIRE_MINUTES = 30

# 安全方案
security = HTTPBearer()


# 模型
class User(BaseModel):
    id: int
    username: str
    email: str


class Token(BaseModel):
    access_token: str
    token_type: str


# 模拟用户数据库
fake_users_db = {
    1: {"id": 1, "username": "admin", "email": "admin@example.com", "password": "admin123"},
    2: {"id": 2, "username": "user", "email": "user@example.com", "password": "user123"},
}


# 依赖：解析并验证 Token
def verify_token(credentials: HTTPAuthorizationCredentials = Depends(security)):
    """验证 JWT Token"""
    token = credentials.credentials
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        user_id = payload.get("sub")
        if user_id is None:
            raise HTTPException(status_code=401, detail="无效的 Token")
        return int(user_id)
    except jwt.ExpiredSignatureError:
        raise HTTPException(status_code=401, detail="Token 已过期")
    except jwt.JWTError:
        raise HTTPException(status_code=401, detail="无效的 Token")


# 依赖：获取当前用户
def get_current_user(user_id: int = Depends(verify_token)):
    """获取当前登录用户"""
    user = fake_users_db.get(user_id)
    if user is None:
        raise HTTPException(status_code=401, detail="用户不存在")
    return User(id=user["id"], username=user["username"], email=user["email"])


# 依赖：检查是否是管理员
def is_admin(current_user: User = Depends(get_current_user)):
    """检查是否是管理员"""
    if current_user.username != "admin":
        raise HTTPException(status_code=403, detail="需要管理员权限")
    return current_user


# 接口：登录获取 Token
@app.post("/login", response_model=Token)
def login(username: str, password: str):
    """登录接口"""
    # 查找用户
    user = None
    for u in fake_users_db.values():
        if u["username"] == username and u["password"] == password:
            user = u
            break
    
    if not user:
        raise HTTPException(status_code=401, detail="用户名或密码错误")
    
    # 生成 Token
    expire = datetime.utcnow() + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
    payload = {"sub": str(user["id"]), "exp": expire}
    token = jwt.encode(payload, SECRET_KEY, algorithm=ALGORITHM)
    
    return {"access_token": token, "token_type": "bearer"}


# 接口：获取当前用户信息（需要登录）
@app.get("/users/me", response_model=User)
def get_me(current_user: User = Depends(get_current_user)):
    """获取当前用户信息"""
    return current_user


# 接口：管理员接口（需要管理员权限）
@app.get("/admin/users")
def get_all_users(admin: User = Depends(is_admin)):
    """获取所有用户（管理员）"""
    return list(fake_users_db.values())
```

## 十一、常见问题

### Q1: 什么时候使用 yield？

当依赖需要清理资源时使用 yield：

```python
def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()  # 请求结束后执行
```

### Q2: 如何在依赖中获取请求对象？

```python
from fastapi import Request

def dependency(request: Request):
    return {"path": request.url.path}
```

### Q3: 如何跳过依赖？

测试时可以使用 `app.dependency_overrides`：

```python
from fastapi.testclient import TestClient

def override_dependency():
    return {"mocked": True}

app.dependency_overrides[get_current_user] = override_dependency

client = TestClient(app)
response = client.get("/users/me")
```
